請將原有的文件複製邏輯改成較有錯誤處理的結構。

javascript 
// ... 上述代碼保持不變...

const changedFiles = [];
let page = 1;

try {
  while (true) {
    const filesResp = await github.rest.pulls.listFiles({
      owner,
      repo,
      pull_number: issue_number,
      per_page: 100,
      page: page
    });

    changedFiles.push(
      ...filesResp.data.map(f => ({
        filename: f.filename,
        status: f.status,
        additions: f.additions,
        deletions: f.deletions,
        patch: f.patch || ""
      }))
    );

    // 檢查是否有足夠的資料，如果少於 100 個，表示這是最後一頁。
    if (filesResp.data.length < 100) {
      break;
    }

    page += 1;
  }
} catch (error) {
  console.error("Error fetching pull request files during pagination:", error);
}

// ... 後續代碼繼續使用 changedFiles 變數...
